//!HOOK LINEAR
//!BIND HOOKED
//!DESC Pixel 9 Pro XL - Perceptual SDR Expansion v3.1 Workstation v6

/*
    Pixel 9 Pro XL - Perceptual SDR Expansion v3.1
    ------------------------------------------------------------

    Designed specifically around the empirically working:
        vo=gpu
        gpu-api=vulkan
        SDR-intensity=4.16
        target-peak=auto
        target-colorspace-hint=no

    The shader runs at mpv's LINEAR stage so luminance work is performed
    in linear light using mpv's own transfer-function conversion.

    Goals:
      1. Preserve true black and white.
      2. Replace global gamma/contrast compensation with a smooth,
         anchored luminance curve.
      3. Preserve 18% reference gray as the main luminance pivot.
      4. Increase upper-mid/highlight luminance without hard clipping.
      5. Measure chroma in Oklab rather than raw RGB distance.
      6. Give bright saturated colors more color-volume expansion than
         ordinary SDR midtones.
      7. Protect near-neutral colors and skin-like hues.
      8. Keep hue approximately constant while increasing chroma.
      9. Gamut-limit BEFORE clipping using a high-quality iterative
         Oklab chroma search.
     10. Never perform a final per-channel clamp as a substitute for
         gamut mapping.

    Notes:
      - Rec.709 and sRGB use the same RGB primaries and D65 white point,
        so the standard linear-sRGB <-> Oklab matrices are valid for
        linear Rec.709 RGB.
      - This is NOT a fake Rec.709 -> BT.2020 matrix conversion.
      - This is SDR perceptual expansion for the special Pixel/vo=gpu
        high-brightness path.
*/

const vec3 LUMA709 = vec3(0.2126, 0.7152, 0.0722);

const float RGB_LOW  = @@RGB_LOW@@;
const float RGB_HIGH = @@RGB_HIGH@@;

/* Workstation-only output/view controls. Compile-time constants are
   regenerated by Lua, so normal DEBUG_VIEW=0 can be optimized away. */
const float SDR_COMPRESS = @@SDR_COMPRESS@@;
const int DEBUG_VIEW = @@DEBUG_VIEW@@;

const float PI = 3.1415926535897932384626433832795;


/* ---------------------------------------------------------
   Oklab conversion
   --------------------------------------------------------- */

vec3 linear709_to_oklab(vec3 c)
{
    // Rec.709 has the same primaries / D65 white as sRGB.
    float l =
        0.4122214708 * c.r +
        0.5363325363 * c.g +
        0.0514459929 * c.b;

    float m =
        0.2119034982 * c.r +
        0.6806995451 * c.g +
        0.1073969566 * c.b;

    float s =
        0.0883024619 * c.r +
        0.2817188376 * c.g +
        0.6299787005 * c.b;

    // Input is expected to be valid non-negative linear RGB.
    float l_ = pow(max(l, 0.0), 1.0 / 3.0);
    float m_ = pow(max(m, 0.0), 1.0 / 3.0);
    float s_ = pow(max(s, 0.0), 1.0 / 3.0);

    return vec3(
         0.2104542553 * l_
       + 0.7936177850 * m_
       - 0.0040720468 * s_,

         1.9779984951 * l_
       - 2.4285922050 * m_
       + 0.4505937099 * s_,

         0.0259040371 * l_
       + 0.7827717662 * m_
       - 0.8086757660 * s_
    );
}


vec3 oklab_to_linear709(vec3 lab)
{
    float l_ =
        lab.x
      + 0.3963377774 * lab.y
      + 0.2158037573 * lab.z;

    float m_ =
        lab.x
      - 0.1055613458 * lab.y
      - 0.0638541728 * lab.z;

    float s_ =
        lab.x
      - 0.0894841775 * lab.y
      - 1.2914855480 * lab.z;

    float l = l_ * l_ * l_;
    float m = m_ * m_ * m_;
    float s = s_ * s_ * s_;

    return vec3(
         4.0767416621 * l
       - 3.3077115913 * m
       + 0.2309699292 * s,

        -1.2684380046 * l
       + 2.6097574011 * m
       - 0.3413193965 * s,

        -0.0041960863 * l
       - 0.7034186147 * m
       + 1.7076147010 * s
    );
}


/* ---------------------------------------------------------
   Helpers
   --------------------------------------------------------- */

bool in_gamut(vec3 c)
{
    return
        c.r >= RGB_LOW  && c.r <= RGB_HIGH &&
        c.g >= RGB_LOW  && c.g <= RGB_HIGH &&
        c.b >= RGB_LOW  && c.b <= RGB_HIGH;
}


float angular_distance(float a, float b)
{
    float d = a - b;
    return abs(atan(sin(d), cos(d)));
}


/*
    Nonlinear luminance expansion.

    Properties:
      y=0.00 -> 0.00
      y=0.18 -> ~0.18
      y=1.00 -> 1.00

    Lower shadows are very slightly deepened.
    Upper midtones/highlights are progressively lifted.
    The curve has no hard knee and no white clipping.
*/
float expand_luminance(float y)
{
    y = clamp(y, 0.0, 1.0);

    const float PIVOT = @@LUMA_PIVOT@@;

    // Smooth S-curve around reference gray.
    float contrastTerm =
        @@LUMA_CONTRAST@@ * (y - PIVOT) * y * (1.0 - y);

    // Additional upper-mid/highlight expansion.
    float highlightGate =
        smoothstep(@@LUMA_HIGHLIGHT_START@@, @@LUMA_HIGHLIGHT_END@@, y);

    float highlightTerm =
        @@LUMA_HIGHLIGHT@@ * highlightGate * y * (1.0 - y); // +7.5% vs v3

    return clamp(
        y + contrastTerm + highlightTerm,
        0.0,
        1.0
    );
}


/*
    Skin-like hue protection in Oklab.

    This intentionally protects only moderate-chroma, human-skin-like
    oranges/reds. Highly saturated orange objects are not strongly
    protected, so the feature does not globally suppress warm colors.
*/
float skin_mask(vec3 lab)
{
    float L = lab.x;
    float C = length(lab.yz);

    if (C < 0.000001)
        return 0.0;

    float hue = atan(lab.z, lab.y);

    // Approximate Oklab skin-family hue center: warm orange-red.
    const float SKIN_CENTER = @@SKIN_CENTER@@; // ~50 degrees

    float hd = angular_distance(hue, SKIN_CENTER);

    float hueMask =
        1.0 - smoothstep(@@SKIN_HUE_INNER@@, @@SKIN_HUE_OUTER@@, hd);

    float lightMask =
        smoothstep(@@SKIN_L_LOW_START@@, @@SKIN_L_LOW_FULL@@, L) *
        (1.0 - smoothstep(@@SKIN_L_HIGH_START@@, @@SKIN_L_HIGH_END@@, L));

    float chromaMask =
        smoothstep(@@SKIN_C_LOW_START@@, @@SKIN_C_LOW_FULL@@, C) *
        (1.0 - smoothstep(@@SKIN_C_HIGH_START@@, @@SKIN_C_HIGH_END@@, C));

    return clamp(hueMask * lightMask * chromaMask, 0.0, 1.0);
}


/*
    High-quality hue-preserving Oklab gamut limiter.

    Starting color is guaranteed to be in gamut. We search only the
    chroma scale from 1.0 to requestedScale while holding Oklab L and
    hue direction constant.

    Seven iterations gives 1/128 resolution across the requested
    chroma interval while remaining inexpensive because this shader
    runs before final display scaling.
*/
float find_gamut_chroma_scale(vec3 lab, float requestedScale)
{
    if (requestedScale <= 1.000001)
        return 1.0;

    vec3 direct =
        oklab_to_linear709(
            vec3(lab.x, lab.yz * requestedScale)
        );

    if (in_gamut(direct))
        return requestedScale;

    float lo = 1.0;
    float hi = requestedScale;

    for (int i = 0; i < @@GAMUT_ITERATIONS@@; i++) {
        float mid = 0.5 * (lo + hi);

        vec3 candidate =
            oklab_to_linear709(
                vec3(lab.x, lab.yz * mid)
            );

        if (in_gamut(candidate))
            lo = mid;
        else
            hi = mid;
    }

    // Keep a tiny safety margin inside the discovered boundary.
    return 1.0 + (lo - 1.0) * @@GAMUT_MARGIN@@;
}


/* ---------------------------------------------------------
   Main pass
   --------------------------------------------------------- */

vec4 hook()
{
    vec4 src = HOOKED_tex(HOOKED_pos);

    // LINEAR hook: mpv has already converted the source to linear light.
    vec3 rgb = max(src.rgb, vec3(0.0));

    float Y = dot(rgb, LUMA709);

    if (Y <= 0.000001)
        return vec4(vec3(0.0), src.a);


    /* -----------------------------------------------------
       1. LUMINANCE / CONTRAST EXPANSION
       ----------------------------------------------------- */

    float targetY = expand_luminance(Y);

    // Chromaticity-preserving linear-light luminance scale.
    float requestedLumScale =
        targetY / max(Y, 0.000001);

    // Prevent luminance expansion itself from clipping a channel.
    float maxChannel =
        max(max(rgb.r, rgb.g), rgb.b);

    float maxLumScale =
        RGB_HIGH / max(maxChannel, 0.000001);

    float lumScale =
        min(requestedLumScale, maxLumScale);

    bool lumaHighLimited =
        requestedLumScale > maxLumScale + 0.000001;

    bool lumaLowLimited =
        targetY <= RGB_LOW && Y > RGB_LOW + 0.000001;

    vec3 expandedRGB =
        rgb * lumScale;

    float expandedY =
        dot(expandedRGB, LUMA709);


    /* -----------------------------------------------------
       2. PERCEPTUAL COLOR-VOLUME EXPANSION
       ----------------------------------------------------- */

    vec3 lab =
        linear709_to_oklab(expandedRGB);

    float L = lab.x;
    float C = length(lab.yz);

    // Oklab chroma relative to perceived lightness.
    float perceptualSaturation =
        C / max(L, @@SAT_L_FLOOR@@);

    // Protect neutral / nearly-neutral material.
    float saturationGate =
        smoothstep(
            @@SAT_GATE_START@@,
            @@SAT_GATE_FULL@@,
            perceptualSaturation
        );

    // Protect deep shadows from chroma/noise amplification.
    float shadowGate =
        smoothstep(
            @@SHADOW_GATE_START@@,
            @@SHADOW_GATE_FULL@@,
            expandedY
        );

    // Separate color-volume behavior for midtones and highlights.
    float midtoneGate =
        smoothstep(@@MIDTONE_START@@, @@MIDTONE_FULL@@, expandedY) *
        (1.0 - smoothstep(@@MIDTONE_FADE_START@@, @@MIDTONE_FADE_END@@, expandedY));

    float highlightGate =
        smoothstep(@@BRIGHT_START@@, @@BRIGHT_FULL@@, expandedY);

    /*
        Midtones receive restrained enhancement.
        Bright colorful material receives substantially more expansion.
    */
    float chromaBoost =
          @@BASE_CHROMA@@
        + @@MID_CHROMA@@ * midtoneGate
        + @@BRIGHT_CHROMA@@ * highlightGate; // +7.5% vs v3

    // Human skin-like colors retain much more of their source chroma.
    float skin =
        skin_mask(lab);

    float skinProtection =
        mix(1.0, @@SKIN_RETAIN@@, skin);

    float requestedChromaScale =
        1.0 +
        chromaBoost *
        saturationGate *
        shadowGate *
        skinProtection;


    /* -----------------------------------------------------
       3. HUE-PRESERVING GAMUT LIMIT
       ----------------------------------------------------- */

    float chromaScale =
        find_gamut_chroma_scale(
            lab,
            requestedChromaScale
        );

    bool gamutLimited =
        chromaScale + 0.000001 < requestedChromaScale;

    vec3 outLab =
        vec3(
            lab.x,
            lab.yz * chromaScale
        );

    vec3 outRGB =
        oklab_to_linear709(outLab);

    /*
        HDR -> SDR compression is deliberately non-destructive.
        0.0 = full current tuning.
        1.0 = original source in this LINEAR hook.
        Intermediate values compress the complete tuned result back toward
        conventional SDR while preserving the current control relationships.
    */
    vec3 finalRGB = outRGB;

    if (SDR_COMPRESS > 0.000001)
        finalRGB = mix(outRGB, rgb, SDR_COMPRESS);

    /*
        Toggleable clipping diagnostics. Non-flagged pixels remain visible
        at low intensity for spatial context.

        GAMUT: magenta = Oklab gamut limiter active.
        LUMA:  red = upper luminance limiter active; blue = lower boundary.
        BOTH:  yellow = gamut + upper-luma, cyan = gamut + lower-luma.
    */
    if (DEBUG_VIEW != 0) {
        vec3 context = finalRGB * 0.16;

        if (DEBUG_VIEW == 1) {
            if (gamutLimited)
                return vec4(1.0, 0.0, 1.0, src.a);
            return vec4(context, src.a);
        }

        if (DEBUG_VIEW == 2) {
            if (lumaHighLimited)
                return vec4(1.0, 0.0, 0.0, src.a);
            if (lumaLowLimited)
                return vec4(0.0, 0.25, 1.0, src.a);
            return vec4(context, src.a);
        }

        if (gamutLimited && lumaHighLimited)
            return vec4(1.0, 1.0, 0.0, src.a);
        if (gamutLimited && lumaLowLimited)
            return vec4(0.0, 1.0, 1.0, src.a);
        if (lumaHighLimited)
            return vec4(1.0, 0.0, 0.0, src.a);
        if (lumaLowLimited)
            return vec4(0.0, 0.25, 1.0, src.a);
        if (gamutLimited)
            return vec4(1.0, 0.0, 1.0, src.a);
        return vec4(context, src.a);
    }

    /*
        No final hard clamp. The luminance limiter and Oklab gamut search
        keep the normal output inside the legal RGB cube.
    */
    return vec4(finalRGB, src.a);
}
